1. REGISTER NEW USER

     "password": "SecurePassword123!"
   }
   
   Response (Success - 200):
   {
     "success": true,
     "message": "Registration successful",
     "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",

       "name": "John Doe",
       "email": "john@example.com"
     }
   }
   
   Response (Error - 400):
   {
     "success": false,
     "message": "User already exists" or "All fields are required"
   }


2. LOGIN USER
   ──────────────────────────────────────────────────────────────────────
   Endpoint: POST /api/auth/login
   Full URL: https://realty-ai-price-persona-predictor.onrender.com/api/auth/login
   
   Headers:
     Content-Type: application/json
   
   Request Body:
   {
     "email": "john@example.com",
     "password": "SecurePassword123!"
   }
   
   Response (Success - 200):
   {
     "success": true,
     "message": "Login successful",
     "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
     "user": {
       "id": "691a0d334c8eecada6686b7d",
       "name": "John Doe",
       "email": "john@example.com"
     }
   }
   
   Response (Error - 400):
   {
     "success": false,
     "message": "Invalid password" or "User not found"
   }


3. VERIFY TOKEN
   ──────────────────────────────────────────────────────────────────────
   Endpoint: GET /api/auth/verify
   Full URL: https://realty-ai-price-persona-predictor.onrender.com/api/auth/verify
   
   Headers:
     Authorization: Bearer <token>
     OR
     Cookie: token=<token>
   
   Response (Success - 200):
   {
     "success": true,
     "user": {
       "id": "691a0d334c8eecada6686b7d",
       "name": "John Doe",
       "email": "john@example.com"
     }
   }
   
   Response (Error - 401):
   {
     "success": false,
     "message": "Not authenticated" or "Invalid token"
   }


4. LOGOUT USER
   ──────────────────────────────────────────────────────────────────────
   Endpoint: POST /api/auth/logout
   Full URL: https://realty-ai-price-persona-predictor.onrender.com/api/auth/logout
   
   Headers:

     OR

   {
     "success": true,
     "message": "Logout successful"
